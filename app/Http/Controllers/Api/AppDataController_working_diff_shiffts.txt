    public function refresh(Request $request)
    {
        $employeeId = $request->input('employee_id');

        if (!$employeeId) {
            return response()->json(['error' => 'employee_id is required'], 400);
        }

        $data = $this->getAppData($employeeId);

        if (!$data) {
            return response()->json(['error' => 'Employee data not found or no assigned shift'], 404);
        }

        return response()->json($data);
    }

    private function getAppData($employeeId)
    {
        $employee = Employee::find($employeeId);
        if (!$employee) {
            return null;
        }

        $employeeShift = EmployeeShift::where('emp_id', $employeeId)->with('shift')->first();
        if (!$employeeShift || !$employeeShift->shift) {
            return null;
        }

        $shiftDate = resolve_shift_date($employeeShift);

        // dd($shiftDate);
        if (!$shiftDate) {
            return [
                'attendance_status' => 'Not Checked In',
                'assigned_shift' => [
                    'shift_name' => $employeeShift->shift->shift_name,
                    'crosses_midnight' => $employeeShift->shift->crosses_midnight,
                    'start_time' => $employeeShift->shift->start_time,
                    'end_time' => $employeeShift->shift->end_time ?? null,
                ],
                'leave_stats' => get_leave_stats($employee),
                'attendance_stats' => [
                    'total_work_days' => 0,
                    'present_days' => 0,
                    'on_time_days' => 0,
                    'late_days' => 0,
                    'on_time_percentage' => 0,
                    'today_status' => 'Not Checked In',
                ],
                'break_stats' => [],
                'app_settings' => get_app_settings(),
                'isOnBreak' => false,
            ];
        }

        $attendance = Attendance::where('emp_id', $employeeId)
            ->where('shift_date', $shiftDate) // keep as plain where
            ->latest()
            ->first();

            // dd($attendance);

        $attendanceStatus = $attendance ? 'Checked In' : 'Not Checked In';

        // Today status
        $todayStatus = 'Not Checked In';
        $lateThreshold = (int) get_app_settings('late_minutes') ?? 5;
        $earlyThreshold = 60;

        if ($attendance && $attendance->check_in) {
            $todayStatus = $this->employeeService->getAttendanceStatus(
                $attendance->check_in,
                $employeeShift->shift->start_time,
                $earlyThreshold,
                $lateThreshold
            );
        }

        // Monthly stats
        $startOfMonth = Carbon::now('Asia/Karachi')->startOfMonth();
        $endOfMonth = Carbon::now('Asia/Karachi')->endOfMonth();

        $workDays = collect(CarbonPeriod::create($startOfMonth, $endOfMonth))
            ->filter(fn($date) => !$date->isWeekend())
            ->count();

        $presentAttendances = Attendance::where('emp_id', $employeeId)
            ->where('status', 'Present')
            ->whereBetween('shift_date', [$startOfMonth, $endOfMonth])
            ->get()
            ->filter(fn($a) => !Carbon::parse($a->shift_date)->isWeekend());

        $presentDays = $presentAttendances->count();

        $lateDays = LateArrival::where('emp_id', $employeeId)
            ->whereBetween('date', [$startOfMonth, $endOfMonth])
            ->get()
            ->filter(fn($l) => !Carbon::parse($l->date)->isWeekend())
            ->count();

        $onTimeDays = max($presentDays - $lateDays, 0);
        $onTimePercentage = $presentDays > 0 ? round(($onTimeDays / $presentDays) * 100) : 0;

        $attendanceStats = [
            'total_work_days' => $workDays,
            'present_days' => $presentDays,
            'on_time_days' => $onTimeDays,
            'late_days' => $lateDays,
            'on_time_percentage' => $onTimePercentage,
            'today_status' => $todayStatus,
        ];

        // Break stats (using same controller logic)
        $breakStats = calculate_break_stats($employee, $employeeShift, $shiftDate);

        // Check ongoing break
        $ongoingBreak = EmployeeBreak::where('emp_id', $employeeId)
            ->where('status', 'On Break')
            ->first();

        return [
            'attendance_status' => $attendanceStatus,
            'assigned_shift' => [
                'shift_name' => $employeeShift->shift->shift_name,
                'crosses_midnight' => $employeeShift->shift->crosses_midnight,
                'start_time' => $employeeShift->shift->start_time,
                'end_time' => $employeeShift->shift->end_time ?? null,
            ],
            'leave_stats' => get_leave_stats($employee),
            'attendance_stats' => $attendanceStats,
            'break_stats' => $breakStats,
            'app_settings' => get_app_settings(),
            'isOnBreak' => $ongoingBreak ? true : false,
        ];
    }

    private function resolveShiftDate($assignedShift)
    {
        $now = Carbon::now('Asia/Karachi');

        $shiftDate = Carbon::parse($assignedShift->shift_date, 'Asia/Karachi');
        $shiftStart = Carbon::parse($shiftDate->toDateString() . ' ' . $assignedShift->start_time, 'Asia/Karachi');
        $shiftEnd = Carbon::parse($shiftDate->toDateString() . ' ' . $assignedShift->end_time, 'Asia/Karachi');

        if ($assignedShift->crosses_midnight && $shiftEnd->lt($shiftStart)) {
            $shiftEnd->addDay();
        }

        if ($now->between($shiftStart, $shiftEnd)) {
            return $shiftDate->toDateString();
        }

        if ($assignedShift->crosses_midnight) {
            $yesterday = $shiftDate->copy()->subDay();
            $yesterdayStart = Carbon::parse($yesterday->toDateString() . ' ' . $assignedShift->start_time, 'Asia/Karachi');
            $yesterdayEnd = Carbon::parse($yesterday->toDateString() . ' ' . $assignedShift->end_time, 'Asia/Karachi');

            if ($yesterdayEnd->lt($yesterdayStart)) {
                $yesterdayEnd->addDay();
            }

            if ($now->between($yesterdayStart, $yesterdayEnd)) {
                return $yesterday->toDateString();
            }
        }

        return null;
    }
